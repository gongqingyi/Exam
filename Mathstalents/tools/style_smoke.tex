\begin{question}{1}{5}
若 \(x+2=5\)，则 \(x=\underline{\hspace{1.5cm}}\)。
\begin{choices}
\choice{\(1\)}
\choice{\(2\)}
\choice{\(3\)}
\choice{\(4\)}
\end{choices}
\answerline
\end{question}
\solutiondivider
\sectionbanner{第一部分}{单选题}
\begin{solution}{1}{C}{4.0}{代数与方程}
\analysis{移项后计算未知数。}
\begin{enumerate}[label=\textbf{Step \arabic*:},leftmargin=*]
\item 两边同时减去 \(2\)；
\item 得 \(x=3\)。
\end{enumerate}
\methodsummary{等式两边同时进行相同运算，等式仍成立。}
\end{solution}
